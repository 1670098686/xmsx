## 二、四层分层架构设计（真实落地）

严格遵循项目规则中的四层分包，**禁止跨层逆向调用**。

```
┌─────────────────────────────────────────────────────────────────┐
│  UI 展示层 (ui/)                                                │
│  - 仅渲染界面、捕获事件                                          │
│  - 调用 business 接口，接收 (code, data, msg) 三元组            │
│  - 耗时操作必须走 QThread（6 个 Worker 类）                       │
│  - 全局状态 GlobalState.selected_lit_id / ThemeManager.current  │
│  - WindowsActivationClickFilter 拦截 WM_MOUSEACTIVATE           │
└───────────────────────────┬─────────────────────────────────────┘
                            │
                            ▼
┌─────────────────────────────────────────────────────────────────┐
│  Business 业务层 (business/)                                    │
│  - 接收 UI 请求，编排业务流程                                    │
│  - 调用 tool_layer（含 ai_client.py）和 data_layer              │
│  - 捕获业务异常、记录日志、写入 operation_log                    │
│  - 统一返回 (code:int, data:any, msg:str)                       │
│  - AI 三通道降级链：file_data → file_id → text → 本地规则       │
└───────────────┬───────────────────────────┬─────────────────────┘
                │                           │
                ▼                           ▼
┌─────────────────────────────┐   ┌─────────────────────────────────┐
│  Tool 工具层 (tool_layer/)   │   │  Data 数据层 (data_layer/)      │
│  - 文件解析、AI 客户端、文本   │   │  - 仅做 SQLite CRUD              │
│  - 分词、关键词提取、路径操作  │   │  - 单例 DatabaseManager         │
│  - 导出文件生成               │   │  - 参数化查询（? 占位符）        │
│  - 零业务逻辑，零 DB 访问     │   │  - 不包含任何业务判断逻辑          │
└─────────────────────────────┘   └─────────────────────────────────┘
```

## 三、数据库设计（SQLite 9 张表，schema v4）

### 3.1 真实 DDL（db_init.py 138 行 DDL + 18 索引 + 4 条迁移）

```sql
PRAGMA foreign_keys = ON;

-- ============================================================
-- schema v1 原始表结构 → v2/v3/v4 增量迁移
-- SCHEMA_VERSION = 4（真实落地，db_init.py 顶部常量）
-- ============================================================

-- 1. category_tag（分类/标签统一表，先建，被 literature_info / literature_tag 关联）
CREATE TABLE IF NOT EXISTS category_tag (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    tag_type     TINYINT      NOT NULL DEFAULT 2,    -- 1=分类目录 2=标签
    tag_name     VARCHAR(50)  NOT NULL,
    tag_color    VARCHAR(20)  NOT NULL DEFAULT '#000000',
    parent_id    INTEGER      NOT NULL DEFAULT 0,
    sort_order   INTEGER      NOT NULL DEFAULT 0,
    create_time  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 2. parse_rule（先建，被 literature_report 关联）
CREATE TABLE IF NOT EXISTS parse_rule (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    rule_name       VARCHAR(50)  NOT NULL,
    subject_type    VARCHAR(20)  NOT NULL DEFAULT '通用',
    rule_detail     TEXT         NOT NULL DEFAULT '{}',
    precision_level TINYINT      NOT NULL DEFAULT 3,
    is_default      TINYINT      NOT NULL DEFAULT 0,
    create_time     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 3. literature_info
CREATE TABLE IF NOT EXISTS literature_info (
    id                 INTEGER PRIMARY KEY AUTOINCREMENT,
    literature_title   VARCHAR(255) NOT NULL DEFAULT '',
    literature_author  VARCHAR(100) NOT NULL DEFAULT '',
    publish_time       VARCHAR(20)  NOT NULL DEFAULT '',
    journal_source     VARCHAR(255) NOT NULL DEFAULT '',
    literature_type    VARCHAR(20)  NOT NULL DEFAULT '',   -- PDF / TXT / DOCX / DOC
    file_path          VARCHAR(500) NOT NULL,              -- 相对路径
    file_size          INTEGER      NOT NULL DEFAULT 0,
    file_hash          VARCHAR(64)  NOT NULL DEFAULT '',
    category_id        INTEGER      NOT NULL DEFAULT 0,
    is_parsed          TINYINT      NOT NULL DEFAULT 0,    -- 0 未解析 1 解析中 2 已解析 3 失败
    create_time        DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    update_time        DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 4. literature_tag（多对多关联表，v1 新增）
CREATE TABLE IF NOT EXISTS literature_tag (
    id             INTEGER PRIMARY KEY AUTOINCREMENT,
    literature_id  INTEGER   NOT NULL,
    tag_id         INTEGER   NOT NULL,
    bind_time      DATETIME  NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (literature_id) REFERENCES literature_info(id) ON DELETE CASCADE,
    FOREIGN KEY (tag_id)        REFERENCES category_tag(id)    ON DELETE CASCADE,
    UNIQUE (literature_id, tag_id)
);

-- 5. literature_report（v3 迁移新增 keywords 列）
CREATE TABLE IF NOT EXISTS literature_report (
    id                    INTEGER PRIMARY KEY AUTOINCREMENT,
    literature_id         INTEGER  NOT NULL,
    research_background   TEXT     NOT NULL DEFAULT '',
    core_view             TEXT     NOT NULL DEFAULT '',
    research_method       TEXT     NOT NULL DEFAULT '',
    innovation_point      TEXT     NOT NULL DEFAULT '',
    research_conclusion   TEXT     NOT NULL DEFAULT '',
    reference_list        TEXT     NOT NULL DEFAULT '',
    keywords              TEXT     NOT NULL DEFAULT '',     -- v3 迁移新增
    parse_time            DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    rule_id               INTEGER  DEFAULT 1,
    FOREIGN KEY (literature_id) REFERENCES literature_info(id) ON DELETE CASCADE,
    FOREIGN KEY (rule_id)       REFERENCES parse_rule(id)      ON DELETE SET NULL
);

-- 6. literature_note（支持 PDF 区域框选 / TXT 段落 / 全局笔记）
CREATE TABLE IF NOT EXISTS literature_note (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    literature_id   INTEGER      NOT NULL,
    paragraph_pos   VARCHAR(50)  NOT NULL DEFAULT '',       -- 锚点：pdf:2:90,170,506,590 / txh:120-186
    note_content    TEXT         NOT NULL DEFAULT '',
    note_type       VARCHAR(20)  NOT NULL DEFAULT 'paragraph',
    highlight_style VARCHAR(30)  NOT NULL DEFAULT '',
    create_time     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    update_time     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (literature_id) REFERENCES literature_info(id) ON DELETE CASCADE
);

-- 7. system_config（键值对表，含 ai_models JSON 数组）
CREATE TABLE IF NOT EXISTS system_config (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    config_key   VARCHAR(100) NOT NULL UNIQUE,
    config_value TEXT         NOT NULL DEFAULT '',
    config_desc  VARCHAR(255) NOT NULL DEFAULT '',
    update_time  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 8. backup_record（全量/增量备份追踪）
CREATE TABLE IF NOT EXISTS backup_record (
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    backup_name   VARCHAR(100) NOT NULL,
    backup_path   VARCHAR(500) NOT NULL,
    backup_size   INTEGER      NOT NULL DEFAULT 0,
    backup_type   VARCHAR(20)  NOT NULL DEFAULT 'full',
    content_scope VARCHAR(100) NOT NULL DEFAULT 'all',
    backup_time   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    backup_status TINYINT      NOT NULL DEFAULT 1
);

-- 9. operation_log（关键操作记录）
CREATE TABLE IF NOT EXISTS operation_log (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    operate_type    VARCHAR(30)  NOT NULL,
    operate_content VARCHAR(500) NOT NULL DEFAULT '',
    literature_id   INTEGER      NOT NULL DEFAULT 0,
    operate_time    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    operate_status  TINYINT      NOT NULL DEFAULT 1
);
```

## 八、单例数据库连接（真实 db_connect.py）

```python
class DatabaseManager:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls, db_path=None):
        with cls._lock:
            if cls._instance is None:
                cls._instance = super().__new__(cls)
                cls._instance._initialize(db_path)
            return cls._instance

    def _initialize(self, db_path):
        # 默认路径 ~/.literature_agent/literature_agent.db
        self.conn = sqlite3.connect(db_path, check_same_thread=False)
        self.conn.row_factory = sqlite3.Row
        self._apply_pragmas()    # foreign_keys=ON / journal_mode=WAL / busy_timeout
```

SCHEMA_VERSION = 4

def init_db(db_manager=None):
    manager = db_manager or DatabaseManager()
    conn = manager.get_conn()
    
    with manager.transaction():
        old_version = conn.execute("PRAGMA user_version").fetchone()[0]
        
        # 所有 DDL + 索引 + 种子数据（幂等，CREATE TABLE IF NOT EXISTS）
        for ddl in _DDL_STATEMENTS:     # 9 张表 DDL
            conn.execute(ddl)
        for index_sql in _INDEX_STATEMENTS:  # 18 索引
            conn.execute(index_sql)
        _seed_parse_rules(conn)         # 4 条种子规则（通用/文科/理科/工科）
        _seed_category_tags(conn)       # 6 条预设分类标签
        _seed_system_config(conn)       # 12 条系统配置（含 ai_models=[]）
        _purge_legacy_keyword_config(conn)  # 清理旧 rule_detail.keyword_top_n
        
        # 版本迁移（幂等：old_version < N 才执行）
        if old_version < 2:
            removed, promoted = _migrate_v1_to_v2_collision_tags(conn)
        if old_version < 3 and _migrate_v2_to_v3_report_keywords(conn):
            pass  # ALTER TABLE ADD keywords
        if old_version < 4:
            doc_count = _migrate_v3_to_v4_doc_type(conn)
        
        conn.execute(f"PRAGMA user_version = {SCHEMA_VERSION}")
